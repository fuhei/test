Usage:  dwarfdump  <options> <object file>
Usage: DwarfDump <options> <object file>
options:	-a	print all .debug_* sections
		-b	print abbrev section
		-c	print loc section
		-c<str>	check only specific compiler objects
		  	  <str> is described by 'DW_AT_producer'. Examples:
		  	    -cg       check only GCC compiler objects
		  	    -cs       check only SNC compiler objects
		  	    -c'350.1' check only compiler objects with 350.1 in the CU name
		-C	activate printing (with -i) of warnings about
			certain common extensions of DWARF.
		-d	dense: one line per entry (info section only)
		-D	do not show offsets
		-e	ellipsis: short names for tags, attrs etc.
		-E[hliaprfoRstxd]	print object Header and/or section information
		  	h=header,l=line,i=info,a=abbrev,p=pubnames,r=aranges,
		  	f=frames,o=loc,R=Ranges,s=strings,t=pubtypes,x=text,
		  	d=default sections, same as -E and {liaprfoRstx}
		-f	print dwarf frame section
		-F	print gnu .eh_frame section
		-g	(use incomplete loclist support)
		-G	show global die offsets
		-h	print IRIX exception tables (unsupported)
		-H <num>	limit output to the first <num> major units
			  example: to stop after <num> compilation units
		-i	print info section
		-I	print sections .gdb_index, .debug_cu_index, .debug_tu_index
		-k[abcdDeEfFgGilmMnrRsStu[f]x[e]y] check dwarf information
		   a	do all checks
		   b	check abbreviations
		   c	examine DWARF constants
		   d	show check results
		   D	check duplicated attributes
		   e	examine attributes of pubnames
		   E	examine attributes encodings
		   f	examine frame information (use with -f or -F)
		   F	examine integrity of files-lines attributes
		   g	check debug info gaps
		   G	print only unique errors
		   i	display summary for all compilers
		   l	check location list (.debug_loc)
		   m	check ranges list (.debug_ranges)
		   M	check ranges list (.debug_aranges)
		   n	examine names in attributes
		   r	examine tag-attr relation
		   R	check forward references to DIEs (declarations)
		   s	perform checks in silent mode
		   S	check self references to DIEs
		   t	examine tag-tag relations
		   u	print tag-tree and tag-attribute usage (basic format)
		   uf	print tag-tree and tag-attribute usage (full format)
		   x	basic frames check (.eh_frame, .debug_frame)
		   xe	extensive frames check (.eh_frame, .debug_frame)
		   y	examine type info
			Unless -C option given certain common tag-attr and tag-tag
			extensions are assumed to be ok (not reported).
		-l[s]	print line section
		   s	do not print <pc> address
		-m	print macinfo section
		-M	print the form name for each attribute
		-n	suppress frame information function name lookup
		  	(when printing frame information from multi-gigabyte
		  	object files this option may save significant time).
		-N	print ranges section
		-O file=<path>	name the output file
		-o[liaprfoR]	print relocation info
		  	l=line,i=info,a=abbrev,p=pubnames,r=aranges,f=frames,o=loc,R=Ranges
		-p	print pubnames section
		-P	print list of compile units per producer
		-Q	suppress printing section data
		-r	print aranges section
		-R	Print frame register names as r33 etc
		  	    and allow up to 1200 registers.
		  	    Print using a 'generic' register set.
		-s	print string section
		-S[v] <option>=<text>	search for <text> in attributes
		  	v	print number of occurrences
		  	with <option>:
		  	-S any=<text>	any <text>
		  	-S match=<text>	matching <text>
		  	-S regex=<text>	use regular expression matching
		  	 (only one -S option allowed, any= and regex= 
		  	  only usable if the functions required are 
		  	  found at configure time)
		-t[afv] static: 
		   a	print both sections
		   f	print static func section
		   v	print static var section
		-u<file> print sections only for specified file
		-v	verbose: show more information
		-vv verbose: show even more information
		-V print version information
		-x abi=<abi>	name abi in dwarfdump.conf
		-x groupnumber=<n>	groupnumber to print
		-x name=<path>	name dwarfdump.conf
		-x tied=<tiedpath>	name an associated object file (Split DWARF)
		-w	print weakname section
		-W	print parent and children tree (wide format) with the -S option
		-Wp	print parent tree (wide format) with the -S option
		-Wc	print children tree (wide format) with the -S option
		-y	print type section
